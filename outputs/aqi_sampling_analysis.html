Visualizing the Sampling Distribution

import matplotlib.pyplot as plt
import seaborn as sns

plt.figure(figsize=(10,5))

sns.histplot(
    estimate_df["estimate"],
    color="darkorange",
    kde=True
)

plt.axvline(
    x=estimate_df["estimate"].mean(),
    color="green",
    label="Mean of Sample Means"
)

plt.axvline(
    x=population_mean,
    color="red",
    label="Population Mean"
)

plt.axvline(
    x=sample_mean1,
    color="black",
    label="First Sample Mean"
)

plt.title("Sampling Distribution of AQI Sample Means")
plt.xlabel("Sample Mean AQI")
plt.ylabel("Density")

plt.grid(True)
plt.legend()
plt.tight_layout()

plt.show()
